1use log::{info, warn};
2use wgpu_quick_start::{my_surface::MySurface, sync_surface::create_new_surface};
3use winit::{event::WindowEvent, window::WindowAttributes};
4use winit_app::{app_listener::AppWindowEvent, application::Application};
5
6fn main() -> Result<(), Box<dyn std::error::Error>> {
7    let winit_app = Application::new();
8    let mut opt_surface: Option<Box<MySurface<'_>>> = None;
9    winit_app.run(
10        WindowAttributes::default().with_title("wgpu quick start app"),
11        move |app_window_event| match app_window_event {
12            AppWindowEvent::NewWindow(window) => match create_new_surface(window) {
13                Ok(value) => {
14                    let boxed = Box::new(value);
15                    opt_surface = Some(boxed);
16                }
17                Err(err) => {
18                    warn!("Error creating new surface from the window {:?}", err);
19                }
20            },
21            AppWindowEvent::OnWindowEvent(event, event_loop) => {
22                if let Some(local_surface) = opt_surface.as_mut() {
23                    match event {
24                        WindowEvent::CloseRequested => {
25                            event_loop.exit();
26                        }
27                        WindowEvent::SurfaceResized(size) => {
28                            info!("Resized {:?}", size);
29                            local_surface.resize((size.width, size.height));
30                        }
31                        WindowEvent::RedrawRequested => match local_surface.get_texture() {
32                            Ok((output, view)) => {
33                                let mut encoder = local_surface.on_device(|device| {
34                                    device.create_command_encoder(&wgpu::CommandEncoderDescriptor {
35                                        label: Some("Render Encoder"),
36                                    })
37                                });
38                                {
39                                    let _render_pass =
40                                        encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
41                                            label: Some("render-pass"),
42                                            color_attachments: &[Some(
43                                                wgpu::RenderPassColorAttachment {
44                                                    view: &view,
45                                                    resolve_target: None,
46                                                    ops: wgpu::Operations {
47                                                        load: wgpu::LoadOp::Clear(wgpu::Color {
48                                                            r: 0.9,
49                                                            g: 0.9,
50                                                            b: 0.9,
51                                                            a: 1.0,
52                                                        }),
53                                                        store: wgpu::StoreOp::Store,
54                                                    },
55                                                    depth_slice: None,
56                                                },
57                                            )],
58                                            depth_stencil_attachment: None,
59                                            occlusion_query_set: None,
60                                            timestamp_writes: None,
61                                            multiview_mask: None,
62                                        });
63                                }
64                                local_surface.submit_to_queue(std::iter::once(encoder.finish()));
65                                output.present();
66                            }
67                            Err(err) => {
68                                warn!("Error retrieving current texture {:?}", err);
69                            }
70                        },
71                        _ => {}
72                    }
73                } else {
74                    warn!("No Window set. Ignoring the window event hence");
75                }
76            }
77        },
78    )?;
79    Ok(())
80}

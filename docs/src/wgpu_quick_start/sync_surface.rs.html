17/// This synchronous function helps create a new surface from the given window.
18///
19/// Available with `sync` feature. (Hence, not available by default).
20///
21/// Can be enabled as below in the crate when being used.
22///
23/// ```toml
24/// wgpu_quick_start = { version="0.28.3", features = ["sync"] }
25/// ```
26///
27/// ## Usage
28///
29/// Following is an example of using `create_new_surface` from a "winit" `Window`.
30///
31///
32/// ```rust
33/// use wgpu_quick_start::{my_surface::MySurface, sync_surface::create_new_surface};
34/// use winit::{event::WindowEvent, window::WindowAttributes};
35/// use winit_app::{app_listener::AppWindowEvent, application::Application};
36///
37///
38/// fn launch() -> Result<(), Box<dyn std::error::Error>> {
39///    let winit_app = Application::new();
40///    let mut opt_surface: Option<Box<MySurface<'_>>> = None;
41///    winit_app.run(
42///        WindowAttributes::default().with_title("wgpu starter app"),
43///        move |app_window_event| match app_window_event {
44///            AppWindowEvent::NewWindow(window) => match create_new_surface(window) {
45///                Ok(value) => {
46///                    opt_surface = Some(Box::new(value));
47///                }
48///                Err(err) => {
49///                    // warning - Error creating new surface from the window
50///                }
51///            },
52///            AppWindowEvent::OnWindowEvent(event, event_loop) => {
53///                if let Some(local_surface) = opt_surface.as_mut() {
54///                     // Handle those events
55///                }
56///            }
57///        },
58///    )?;
59///    Ok(())
60/// }
61/// ```
62#[cfg(feature = "sync")]
63pub fn create_new_surface<'a>(window: Box<dyn Window>) -> GPUStarterResult<MySurface<'a>> {
64    let size = window.surface_size();
65    let surface = pollster::block_on(MySurface::new(window, (size.width, size.height)))?;
66    info!("Created a new surface context. Passing down");
67    Ok(surface)
68}

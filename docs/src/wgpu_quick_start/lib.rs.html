3//! A starter crate to get started with wgpu
4//!
5//! ```rust
6//! use wgpu_quick_start::{my_surface::MySurface, sync_surface::create_new_surface};
7//! use winit::{event::WindowEvent, window::WindowAttributes};
8//! use winit_app::{app_listener::AppWindowEvent, application::Application};
9//!
10//!
11//! fn launch() -> Result<(), Box<dyn std::error::Error>> {
12//!    let winit_app = Application::new();
13//!    let mut opt_surface: Option<Box<MySurface<'_>>> = None;
14//!    winit_app.run(
15//!        WindowAttributes::default().with_title("wgpu starter app"),
16//!        move |app_window_event| match app_window_event {
17//!            AppWindowEvent::NewWindow(window) => match create_new_surface(window) {
18//!                Ok(value) => {
19//!                    let boxed = Box::new(value);
20//!                    opt_surface = Some(boxed);
21//!                }
22//!                Err(err) => {
23//!                    // warning - Error creating new surface from the window
24//!                }
25//!            },
26//!            AppWindowEvent::OnWindowEvent(event, event_loop) => {
27//!                if let Some(local_surface) = opt_surface.as_mut() {
28//!                    match event {
29//!                        WindowEvent::CloseRequested => {
30//!                            event_loop.exit();
31//!                        }
32//!                        WindowEvent::SurfaceResized(size) => {
33//!                            // Resized
34//!                            local_surface.resize((size.width, size.height));
35//!                        }
36//!                        WindowEvent::RedrawRequested => {
37//!                            match local_surface.get_texture() {
38//!                                Ok((output, view)) => {
39//!                                     let mut encoder = local_surface.on_device(|device| {
40//!                                        device.create_command_encoder(
41//!                                            &wgpu::CommandEncoderDescriptor {
42//!                                                label: Some("Render Encoder"),
43//!                                            },
44//!                                        )
45//!                                    });
46//!                                    {
47//!                                        let _render_pass = encoder.begin_render_pass(
48//!                                            &wgpu::RenderPassDescriptor {
49//!                                                label: Some("render-pass"),
50//!                                                color_attachments: &[Some(
51//!                                                    wgpu::RenderPassColorAttachment {
52//!                                                        view: &view,
53//!                                                        resolve_target: None,
54//!                                                        ops: wgpu::Operations {
55//!                                                            load: wgpu::LoadOp::Clear(
56//!                                                                wgpu::Color {
57//!                                                                    r: 0.2,
58//!                                                                    g: 0.2,
59//!                                                                    b: 0.2,
60//!                                                                    a: 1.0,
61//!                                                                },
62//!                                                            ),
63//!                                                            store: wgpu::StoreOp::Store,
64//!                                                        },
65//!                                                        depth_slice: None,
66//!                                                    },
67//!                                                )],
68//!                                                depth_stencil_attachment: None,
69//!                                                occlusion_query_set: None,
70//!                                                timestamp_writes: None,
71//!                                                multiview_mask: None,
72//!                                            },
73//!                                        );
74//!                                    }
75//!                                    local_surface.submit_to_queue(std::iter::once(encoder.finish()));
76//!                                    output.present();
77//!                                }
78//!                                Err(err) => {
79//!                                    // warning - error creating the texture
80//!                                }
81//!                            }
82//!                        }
83//!                        _ => {}
84//!                    }
85//!                }
86//!            }
87//!        },
88//!    )?;
89//!    Ok(())
90//! }
91//! ```

100use thiserror::Error;

102#[derive(Error, Debug)]
103pub enum GPUStarterError {
104    #[error(transparent)]
105    SurfaceError(#[from] wgpu::SurfaceError),
106    #[error(transparent)]
107    CreateSurfaceError(#[from] wgpu::CreateSurfaceError),
108    #[error(transparent)]
109    RequestAdapterError(#[from] wgpu::RequestAdapterError),
110    #[error(transparent)]
111    RequestDeviceError(#[from] wgpu::RequestDeviceError),
112}
113
114pub type GPUStarterResult<T> = Result<T, GPUStarterError>;
